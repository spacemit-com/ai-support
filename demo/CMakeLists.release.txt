cmake_minimum_required(VERSION 3.10)

project(bianbuai-demo)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fpermissive")

# TODO: update cc files with '#ifndef NDEBUG'
set(CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG} -DDEBUG")

# To find OpenCV, you may need to set OpenCV_DIR variable to the
# absolute path to the directory containing OpenCVConfig.cmake file.
find_package(OpenCV PATHS ${BIANBUAI_HOME}/lib/3rdparty/opencv4/lib/cmake/opencv4)

include_directories(${OpenCV_INCLUDE_DIRS} ${BIANBUAI_HOME}/include/bianbuai)

set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -Wl,-rpath-link,${ORT_HOME}/lib:${BIANBUAI_HOME}/lib:${BIANBUAI_HOME}/lib/3rdparty/opencv4/lib")
set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -Wl,--as-needed")
link_directories(${BIANBUAI_HOME}/lib ${BIANBUAI_HOME}/lib/3rdparty/opencv4/lib)

add_executable(classification_demo image_classification_demo.cc)
target_link_libraries(classification_demo PUBLIC bianbuai opencv_core opencv_imgcodecs)


add_executable(detection_demo object_detection_demo.cc)
target_link_libraries(detection_demo PUBLIC bianbuai opencv_core opencv_imgproc opencv_imgcodecs)

install(TARGETS detection_demo classification_demo
        RUNTIME DESTINATION bin
        LIBRARY DESTINATION lib
        ARCHIVE DESTINATION lib)
